LEGENDA
-----------------------------------------------------------------
  IREG        = registrador inteiro (ex: R1, R2, ...).
  FREG        = registrador de ponto flutuante (ex: F1, F2, ...).
  MREG        = registradores HI/LO (sem acesso direto, apenas
                como fonte implícita de instruções
                especializadas).
  IMM         = valor imediato (constante numerica que, no livro,
                sempre começa com um '#').
  IMMU        = valor imediato "unsigned".
  OFF         = valor imediato sem '#' (utilizado no offset de
                instruções de memória).
  LABEL       = rotulo / label de destino (endereco simbolico)
  IMM(IREG)   = modo de enderecamento base+offset: um imediato de
                deslocamento seguido do registrador base entre
                parenteses (ex: 45(R3) = endereco R3 + 45).

  "a/b"       = a barra indica que a posicao aceita mais de um
                tipo/sintaxe (ex: IREG/IMM = aceita registrador
                OU imediato nessa posicao).

  Quando o mesmo opcode tem sintaxes estruturalmente muito
  diferentes (numero ou natureza de operandos distintos), cada
  variante e listada em uma linha separada, ja que sao efetivamente
  instrucoes diferentes por baixo do mesmo nome.

  Comentarios de observacao comportamental vem apos "#" e nao fazem
  parte da sintaxe - servem apenas de nota para a implementacao.


LOADS
-----------------------------------------------------------------
# IREG/FREG, OFF(IREG)
load     IREG/FREG, OFF(IREG)     # generico: destino pode ser int ou float

# IREG, OFF(IREG)
lw       IREG, OFF(IREG)
lb       IREG, OFF(IREG)
lh       IREG, OFF(IREG)
lbu      IREG, OFF(IREG)          # variante unsigned de lb
lhu      IREG, OFF(IREG)          # variante unsigned de lh
ld       IREG, OFF(IREG)          # load doubleword
lwu      IREG, OFF(IREG)          # load word unsigned
ll       IREG, OFF(IREG)          # load linked (uso atomico)

# FREG, OFF(IREG)
l.d      FREG, OFF(IREG)          # destino float; base de endereco sempre int
l.s      FREG, OFF(IREG)


STORES
-----------------------------------------------------------------
# IREG/FREG, OFF(IREG)
store    IREG/FREG, OFF(IREG)     # generico: fonte pode ser int ou float

# IREG, OFF(IREG)
sw       IREG, OFF(IREG)
sb       IREG, OFF(IREG)
sh       IREG, OFF(IREG)
sd       IREG, OFF(IREG)          # store doubleword
sc       IREG, OFF(IREG)          # store conditional (uso atomico)

# FREG, OFF(IREG)
s.d      FREG, OFF(IREG)          # fonte float; base de endereco sempre int
s.s      FREG, OFF(IREG)


INT_BASIC
-----------------------------------------------------------------
# IREG, IREG, IREG
add      IREG, IREG, IREG         # rd, rs, rt
addu     IREG, IREG, IREG
daddu    IREG, IREG, IREG         # versao doubleword de addu
sub      IREG, IREG, IREG
subu     IREG, IREG, IREG
dsubu    IREG, IREG, IREG
and      IREG, IREG, IREG
or       IREG, IREG, IREG
xor      IREG, IREG, IREG
nor      IREG, IREG, IREG
sllv     IREG, IREG, IREG         # rd, rt, rs (shift amount em registrador)
srlv     IREG, IREG, IREG
srav     IREG, IREG, IREG
slt      IREG, IREG, IREG         # rd, rs, rt
sltu     IREG, IREG, IREG
dslt     IREG, IREG, IREG
dsltu    IREG, IREG, IREG

# IREG, IREG, IMM
addi     IREG, IREG, IMM          # rt, rs, imm
addiu    IREG, IREG, IMM
daddiu   IREG, IREG, IMM          # versao doubleword de addiu
subi     IREG, IREG, IMM
slti     IREG, IREG, IMM          # rt, rs, imm
sltiu    IREG, IREG, IMM
dslti    IREG, IREG, IMM
dsltiu   IREG, IREG, IMM

# IREG, IREG, IMMU
andi     IREG, IREG, IMMU
ori      IREG, IREG, IMMU
xori     IREG, IREG, IMMU
sll      IREG, IREG, IMMU          # rd, rt, shamt (shift amount imediato)
srl      IREG, IREG, IMMU
sra      IREG, IREG, IMMU
dsll     IREG, IREG, IMMU          # versao doubleword de sll
dsrl     IREG, IREG, IMMU
dsra     IREG, IREG, IMMU

# IREG, IMMU
lui      IREG, IMMU                # rt, imm - SEM registrador rs

# IREG
mflo     IREG + lo/lo              # rd - fonte implícita: LO (retira o dado do registrador LO - mult).
mfhi     IREG + hi/lo              # rd - fonte implícita: HI (retira o dado do registrador HI - mult).


BRANCHES
-----------------------------------------------------------------
# IREG, IREG, LABEL
beq      IREG, IREG, LABEL
bne      IREG, IREG, LABEL

# IREG, LABEL
bnez     IREG, LABEL              # compara com zero implicito
beqz     IREG, LABEL
bgtz     IREG, LABEL
bltz     IREG, LABEL
bgez     IREG, LABEL
blez     IREG, LABEL
bltzal   IREG, LABEL + ra         # tambem escreve endereco de retorno em "ra"
bgezal   IREG, LABEL + ra         # idem

# IREG, IREG
jalr     IREG, IREG               # rd, rs explicitos

# IREG
jalr     IREG + ra                # rs apenas; rd = $ra implicito
jr       IREG                     # so registrador, sem label

# LABEL
jal      LABEL + ra               # sem registrador; escreve $ra implicitamente
j        LABEL                    # sem registrador


INT_MUL
-----------------------------------------------------------------
# IREG, IREG, IREG
mul      IREG, IREG, IREG         # rd, rs, rt

# IREG, IREG
mult     IREG, IREG + hi/lo       # rs, rt -> resultado (produto) em HI/LO implicito
multu    IREG, IREG + hi/lo       # idem, sem sinal
dmult    IREG, IREG + hi/lo       # versao doubleword de mult
dmultu   IREG, IREG + hi/lo


INT_DIV
-----------------------------------------------------------------
# IREG, IREG
div      IREG, IREG + hi/lo       # rs, rt -> quociente/resto em HI/LO implicito
divu     IREG, IREG + hi/lo
ddiv     IREG, IREG + hi/lo       # versao doubleword de div
ddivu    IREG, IREG + hi/lo

cvt.w.d  FREG, FREG               # double -> word
